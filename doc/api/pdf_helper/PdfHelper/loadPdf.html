Future<void> loadPdf(Uint8List bytes, GameState gameState) async {
  try {
    _ensureEngine();
    _document?.dispose();
    _bytes = bytes;
    _document = await PdfDocument.openData(bytes);
    await _renderPage(0, gameState);
  } catch (e) {
    debugPrint('PDF load error: $e');
  }
}
